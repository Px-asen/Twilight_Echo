if(MSVC)
  target_compile_options(twilight_automix_native PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/fp:strict;/experimental:c11atomics>")
else()
  target_compile_options(twilight_automix_native PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off;-fno-fast-math>")
  if(ANDROID)
    # Inference's portable C kernels must remain usable in debug APKs too.
    # Retain strict FP semantics and debug symbols; do not enable fast-math.
    target_compile_options(twilight_automix_native PRIVATE -O2)
    # The same descriptor identity and bounded reader must work above 2 GiB
    # on 32-bit Android as well as arm64/x86_64.
    target_compile_definitions(twilight_automix_native PUBLIC _FILE_OFFSET_BITS=64)
  endif()
endif()
