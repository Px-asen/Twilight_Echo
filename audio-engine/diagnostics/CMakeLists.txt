cmake_minimum_required(VERSION 3.25)
project(TwilightCrashCapture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(twilight_crash_helper CrashHelper.cpp)
target_compile_definitions(twilight_crash_helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(twilight_crash_helper PRIVATE /Zi /O2)
target_link_options(twilight_crash_helper PRIVATE /DEBUG)
target_link_libraries(twilight_crash_helper PRIVATE dbghelp version)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set_target_properties(twilight_crash_helper PROPERTIES OUTPUT_NAME twilight-crash-helper-x86)
else()
  set_target_properties(twilight_crash_helper PROPERTIES OUTPUT_NAME twilight-crash-helper)
endif()
add_executable(twilight_crash_fixture CrashFixture.cpp)
set_target_properties(twilight_crash_fixture PROPERTIES OUTPUT_NAME twilight-crash-fixture)
add_library(twilight_crash_addon MODULE CrashAddonFixture.cpp ../napi/node_api_dynamic_win32.cpp)
target_include_directories(twilight_crash_addon PRIVATE ../../node_modules/node-api-headers/include)
set_target_properties(twilight_crash_addon PROPERTIES PREFIX "" SUFFIX ".node")
