cmake_minimum_required(VERSION 3.25)
project(TwilightServerService LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The SCM host is a Windows target")
endif()
add_library(twilight_server_credentials MODULE credentials_node.cpp ../../audio-engine/napi/node_api_dynamic_win32.cpp)
target_include_directories(twilight_server_credentials PRIVATE ../../node_modules/node-api-headers/include)
target_compile_features(twilight_server_credentials PRIVATE cxx_std_20)
target_compile_definitions(twilight_server_credentials PRIVATE NAPI_VERSION=10 UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(twilight_server_credentials PRIVATE crypt32)
set_target_properties(twilight_server_credentials PROPERTIES PREFIX "" SUFFIX ".node")
if(MINGW)
  target_link_options(twilight_server_credentials PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_executable(twilight-server-service service_host.cpp)
target_compile_features(twilight-server-service PRIVATE cxx_std_20)
target_compile_definitions(twilight-server-service PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(twilight-server-service PRIVATE advapi32)
if(MINGW)
  target_link_options(twilight-server-service PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_compile_options(twilight-server-service PRIVATE -Wall -Wextra -Werror)
elseif(MSVC)
  target_compile_options(twilight-server-service PRIVATE /W4 /utf-8)
  set_property(TARGET twilight-server-service PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
